1. Бюджет поверок (руб.):
Бюджет_Поверок = SUM('Перечни_из_цехов'[Цена_поверки_руб.])

2. Потенциальный финансовый риск цеха (руб.):
Потенциальный_Риск_Цеха = SUM('Перечни_из_цехов'[Убыток_при_аварии_руб.])

3. Реальный накопленный убыток от просрочки (руб.):
Реальный_Убыток_Просрочки = SUM('Перечни_из_цехов'[Убыток_руб])

4. Риск срыва SLA-поставок в цехе Фасовки (руб.):
Риск_SLA_Поставок = CALCULATE(
    SUM('Перечни_из_цехов'[Убыток_при_аварии_руб.]), 
    'Перечни_из_цехов'[Цех] = "Фасовка"
)
